# DRUMix — a native VST3 MIDI drum sampler for Linux (no framework).
#
# The editor is a Steinberg IPlugView embedded into the host's X11 window and
# painted with Cairo/FreeType — no plug-in or GUI framework, which is what lets
# DRUMix stay MIT-licensed.
#
# Builds two ways:
#   - standalone: point VST3_SDK_DIR at a VST 3 SDK checkout (default: the
#     in-tree vst3sdk/ submodule) and configure this directory directly;
#   - as a subdirectory of a project that already builds that SDK (the parent
#     must provide the SDK targets and public_sdk_SOURCE_DIR).

cmake_minimum_required(VERSION 3.25.0)

project(smtg-drumix
    VERSION 0.1.0
    DESCRIPTION "DRUMix MIDI drum sampler VST3 for Linux"
)

if(NOT TARGET sdk)
    set(VST3_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vst3sdk"
        CACHE PATH "VST 3 SDK source tree")
    if(NOT EXISTS ${VST3_SDK_DIR}/CMakeLists.txt)
        message(FATAL_ERROR
            "VST 3 SDK not found at '${VST3_SDK_DIR}' - run "
            "'git submodule update --init' or set -DVST3_SDK_DIR=<path>")
    endif()

    # Headless SDK: core libraries plus the validator (part of the hosting
    # examples), no VSTGUI (we draw the editor ourselves), no plug-in examples.
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "editor is drawn directly" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "validator" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "plug-in examples not needed" FORCE)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "run manually instead" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "install manually instead" FORCE)

    add_subdirectory(${VST3_SDK_DIR} vst3sdk)

    # The SDK's plugin helper functions read these directory-scoped variables
    # (smtg_target_add_library_main silently adds no module entry point without
    # public_sdk_SOURCE_DIR); the SDK sets them only in its own scope.
    set(public_sdk_SOURCE_DIR "${VST3_SDK_DIR}/public.sdk")
    set(pluginterfaces_SOURCE_DIR "${VST3_SDK_DIR}/pluginterfaces")
endif()

set(DRUMIX_ROOT ${CMAKE_CURRENT_SOURCE_DIR})

# ---------------------------------------------------------------------------
# Editor graphics stack: Cairo drawing, FreeType text and PNG layers. Kept as
# its own library so the offline render tool below can exercise exactly the
# code the editor paints with, without X11 or VST3.
# ---------------------------------------------------------------------------
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo cairo-ft freetype2 fontconfig)
# The editor embeds into the host's X11 window; cairo-xlib paints it.
pkg_check_modules(CAIROX11 REQUIRED IMPORTED_TARGET cairo-xlib x11)

add_library(DrumGfx STATIC
    ${DRUMIX_ROOT}/src/gfx/canvas.cpp
    ${DRUMIX_ROOT}/src/gfx/fontstack.cpp
    ${DRUMIX_ROOT}/src/gfx/image.cpp
    ${DRUMIX_ROOT}/src/drumpanel.cpp
    ${DRUMIX_ROOT}/src/filebrowser.cpp
    ${DRUMIX_ROOT}/src/platform/respath.cpp
)
target_include_directories(DrumGfx PUBLIC ${DRUMIX_ROOT}/src)
target_link_libraries(DrumGfx PUBLIC PkgConfig::CAIRO ${CMAKE_DL_LIBS})
target_compile_features(DrumGfx PUBLIC cxx_std_17)
set_target_properties(DrumGfx PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(DrumGfx PRIVATE -Wall)

# ---------------------------------------------------------------------------
# The plug-in
# ---------------------------------------------------------------------------
smtg_add_vst3plugin(DRUMix
    src/idrumloader.h
    src/drumids.h
    src/drumgeometry.h
    src/version.h
    src/wavsample.h
    src/wavsample.cpp
    src/drumvoices.h
    src/drumvoices.cpp
    src/drumprocessor.h
    src/drumprocessor.cpp
    src/drumcontroller.h
    src/drumcontroller.cpp
    src/drumview.h
    src/drumview.cpp
    src/drumpanel.h
    src/filebrowser.h
    src/platform/x11plugview.h
    src/platform/x11plugview.cpp
    src/drumfactory.cpp
)

smtg_target_configure_version_file(DRUMix)

# Editor sprite layers (generated by gui/make_*.sh, committed) -> bundle
# Contents/Resources/img/.
file(GLOB DRUMIX_IMG CONFIGURE_DEPENDS "${DRUMIX_ROOT}/resources/img/*.png")
smtg_target_add_plugin_resources(DRUMix
    RESOURCES ${DRUMIX_IMG}
    OUTPUT_SUBDIRECTORY img
)

# Title (Michroma) and body (Roboto) faces, loaded straight out of the bundle
# by FreeType — no font installation step.
smtg_target_add_plugin_resources(DRUMix
    RESOURCES
        "${DRUMIX_ROOT}/resources/fonts/Michroma-Regular.ttf"
        "${DRUMIX_ROOT}/resources/fonts/Roboto-Regular.ttf"
    OUTPUT_SUBDIRECTORY fonts
)

target_include_directories(DRUMix PRIVATE ${DRUMIX_ROOT}/src)
target_compile_features(DRUMix PUBLIC cxx_std_17)
target_compile_options(DRUMix PRIVATE -ffast-math -Wall)

target_link_libraries(DRUMix
    PRIVATE
        sdk
        DrumGfx
        PkgConfig::CAIROX11
)

# ---------------------------------------------------------------------------
# panelrender — offline render of the editor to a PNG. No X11, no host, no
# VST3 objects: the graphics stack's test rig, and how GUI changes are
# reviewed alongside a real host load.
# ---------------------------------------------------------------------------
add_executable(panelrender ${DRUMIX_ROOT}/tools/panelrender.cpp)
target_link_libraries(panelrender PRIVATE DrumGfx)
target_compile_options(panelrender PRIVATE -Wall)

# ---------------------------------------------------------------------------
# drumix-standalone — run the plug-in without a DAW: one X window, the plug-in's
# own editor embedded in it, a Linux::IRunLoop the editor registers with, and a
# JACK client giving the processor MIDI in and taking its audio out. The bundle
# is loaded as a module (not linked in) so the editor's dladdr-based resource
# lookup behaves exactly as it does inside a host.
# ---------------------------------------------------------------------------
pkg_check_modules(JACK IMPORTED_TARGET jack)
if(JACK_FOUND)
    # The module loader, the plug-in provider and the host-side interface IIDs
    # are compiled straight in, exactly as the SDK's own hosting samples do.
    add_executable(drumix-standalone
        ${DRUMIX_ROOT}/standalone/main.cpp
        ${DRUMIX_ROOT}/standalone/runloop.h
        ${DRUMIX_ROOT}/standalone/runloop.cpp
        ${DRUMIX_ROOT}/standalone/jackclient.h
        ${DRUMIX_ROOT}/standalone/jackclient.cpp
        ${VST3_SDK_DIR}/public.sdk/source/vst/hosting/module_linux.cpp
        ${VST3_SDK_DIR}/public.sdk/source/vst/hosting/plugprovider.cpp
        ${VST3_SDK_DIR}/public.sdk/samples/vst-hosting/editorhost/source/usediids.cpp
    )
    target_include_directories(drumix-standalone PRIVATE ${DRUMIX_ROOT}/src)
    target_link_libraries(drumix-standalone PRIVATE sdk_hosting PkgConfig::JACK
                                                    PkgConfig::CAIROX11)
    target_compile_features(drumix-standalone PRIVATE cxx_std_17)
    target_compile_options(drumix-standalone PRIVATE -Wall)
    add_dependencies(drumix-standalone DRUMix)
else()
    message(STATUS "[DRUMix] JACK not found - skipping the standalone build")
endif()
